"""Numeric behavioral fixtures; no synthetic research text or model execution."""
import sys
from pathlib import Path
import unittest
import numpy as np
from scipy.stats import shapiro

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
import phase_d_pilot as d


class SequentialStatisticsTests(unittest.TestCase):
    def test_rolling_uses_current_and_past_window_only(self):
        values=np.random.default_rng(45).normal(size=255)
        steps,scores=d.rolling(values,32)
        self.assertEqual(steps[0],102)
        expected=1-shapiro(d.original.base.trim_outliers_3sigma(values[69:101])).statistic
        self.assertAlmostEqual(scores[0],expected)
        changed=values.copy(); changed[101:]+=100
        self.assertEqual(d.rolling(changed,32)[1][0],scores[0])

    def test_cusum_directions_and_two_sided_max(self):
        up,low=d.cusum(np.array([2.,2.,-3.,-3.]),.5)
        np.testing.assert_allclose(up,[1.5,3,0,0])
        np.testing.assert_allclose(low,[0,0,2.5,5])
        np.testing.assert_allclose(np.maximum(up,low),[1.5,3,2.5,5])

    def test_cusum_state_resets_between_streams(self):
        d.cusum(np.ones(5)*10,.5)
        np.testing.assert_array_equal(d.cusum(np.zeros(3),.5)[0],np.zeros(3))

    def test_invalid_and_constant_rolling_fail(self):
        for values in [np.ones(255),np.full(255,np.nan)]:
            with self.assertRaises(ValueError): d.rolling(values,32)

    def test_position_fit_uses_only_supplied_fit_rows_and_floor(self):
        fit=d.fit_distance([np.ones(255),np.ones(255)*1.02],.1)
        np.testing.assert_allclose(fit['mu'],1.01)
        np.testing.assert_allclose(fit['sigma'],.1)

    def test_insufficient_five_percent_calibration_is_infinite(self):
        h=d.calibrate(list(range(8)),.05)
        self.assertFalse(h['finite']); self.assertIsNone(h['threshold'])
        self.assertEqual(h['rank'],9)
        self.assertFalse(d.crossing([102],[999],h)['any_alert'])

    def test_rank_and_tie_are_conservative(self):
        h=d.calibrate([1]*8,.2)
        self.assertEqual(h['rank'],8)
        self.assertFalse(d.crossing([102],[1],h)['any_alert'])
        self.assertTrue(d.crossing([102],[1.01],h)['any_alert'])

    def test_calibration_uses_stream_maximum(self):
        self.assertEqual(d.maximum(([102,103],[.1,.9])),.9)
        self.assertEqual(d.calibrate(range(8),.4)['threshold'],5)

    def test_prefix_alarm_is_not_tp_and_delay_is_null(self):
        row=d.crossing([102,111,120],[2,3,4],{'finite':True,'threshold':1})
        self.assertTrue(row['prefix_alarm']); self.assertFalse(row['eligible'])
        self.assertFalse(row['detected']); self.assertIsNone(row['delay'])

    def test_onset_detection_has_zero_delay(self):
        row=d.crossing([102,111],[0,2],{'finite':True,'threshold':1})
        self.assertTrue(row['detected']); self.assertEqual(row['delay'],0)

    def test_no_detection_is_right_censored(self):
        row=d.crossing([102,256],[0,0],{'finite':True,'threshold':1})
        self.assertTrue(row['right_censored']); self.assertIsNone(row['delay'])

    def test_benign_has_no_detection_or_censor_label(self):
        row=d.crossing([102],[2],{'finite':True,'threshold':1},has_attack=False)
        for field in ['eligible','detected','delay','right_censored']:
            self.assertIsNone(row[field])

    def test_evaluation_cohort_is_rejected(self):
        with self.assertRaises(ValueError): d.check_cohort(15)

    def test_matching_requires_control_vector_and_no_prefix_alarms(self):
        self.assertFalse(d.match_screen([False,False,False],[True,False,False],0,0))
        self.assertFalse(d.match_screen([False]*3,[False]*3,0,1))
        self.assertTrue(d.match_screen([False]*3,[False]*3,0,0))


if __name__=='__main__': unittest.main()
